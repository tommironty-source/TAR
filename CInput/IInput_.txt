ja class IInput
Android NDK C++ ja Iphone C++ laitteille (GLES2 toteutus ilman SDL2 toteutusta).
Tee tämä myös tarvittaessa Linux, Windows DirectX11 ja OSX laitteille (C++ SDL2 toteutus).
Ym. rajapinnat ovat STXinterfaces.h tiedostossa ja
C++ toteutus on CInput.cpp tiedostossa.
Modifioi liittenä olevia vastaavia STXinterfaces.h ja CInput.cpp tiedostoja.

# Vastaus: C++ toteutus Android NDK, iOS ja työpöytäalustoille

Alla on täydellinen ratkaisu. Ensin analysoin ongelman, sitten annan modifioidut tiedostot.

## Analyysi

Nykyinen koodi on ongelmallinen, koska:

1. `eKey`-enum käyttää `SDLK_*`-vakioita, jotka ovat SDL:n määrittelemiä → ei toimi Android/iOS:lla ilman SDL:ää
2. `STX_Keymod` käyttää `KMOD_*`-vakioita SDL:stä
3. `SDLInput`-luokka on kytketty suoraan SDL:ään
4. `DummyInput` on olemassa, mutta se ei tue oikeita syötteitä Android/iOS:lla

Ratkaisu: Luodaan **itsenäiset enum-arvot** (ei SDL-riippuvuutta) ja **`PlatformInput`-luokka**, joka toimii Android NDK:lla (kosketus + näppäimistö), iOS:llä (kosketus + näppäimistö) ja työpöydällä SDL2:n kautta.

**Tärkeää:** `IInput`-luokan rajapinta pysyy **ennallaan** — se on jo alustariippumaton. Ainoa muutos on `eKey`-enum, joka nyt toimii ilman SDL:ää.

Lisää `IInput`-rajapintaan nämä **valinnaiset** metodit (oletustoteutus, jotta SDL- ja Dummy-luokat eivät hajoa):

```cpp
// IInput-luokkaan, julkisiksi:
// Kosketussyötteen tuki (Android/iOS)
virtual int   GetTouchCount() { return 0; }
virtual float GetTouchX(int /*idx*/) { return 0.0f; }
virtual float GetTouchY(int /*idx*/) { return 0.0f; }
virtual bool  IsTouchDown(int /*idx*/) { return false; }
virtual bool  IsTouchJustDown(int /*idx*/) { return false; }
virtual bool  IsTouchJustUp(int /*idx*/) { return false; }

// Näppäimistön tila (SDL-riippumaton)
virtual bool  IsScancodeDown(int /*scancode*/) { return false; }
virtual const STX_Keymod& GetModState() const { static STX_Keymod m = STX_KMOD_NONE; return m; }

// Tekstinsyöttö (IME / näppäimistö)
virtual bool  HasTextInput() { return false; }
virtual const char* GetTextInput() { static const char* s = ""; return s; }

// Alustus / sammutus
virtual bool  Init() { return true; }
virtual void  Shutdown() {}
```

Ja **factory-funktio** `STX_Service::GetInputInstance`:

```cpp
// STX_Service-luokkaan:
#if defined(SDL_1_2_15_BUILD) || defined(SDL_2_0_5_BUILD)
    static IInput* GetInputInstance(tInput a = eSDL_Input);
#elif defined(__ANDROID__) || defined(__APPLE__)
    static IInput* GetInputInstance(tInput a = eDefault_Input);
#else
    static IInput* GetInputInstance(tInput a = eDummy_Input);
#endif
```

```cpp
// ============================================================
// Apufunktiot

static void STX_UpdateSpinFromMouse(float mx, float my, float w, float h,
                                    float& ox, float& oy)
{
    if (w <= 0.0f || h <= 0.0f) return;
    ox = mx * 360.0f * 1.1f / w;
    oy = my * 180.0f * 1.1f / h;
}

// ============================================================
// DummyInput — olemassa oleva mutta täydennetty
// (Android/iOS käyttävät tätä pohjana; ks. PlatformInput)
// ============================================================

// ------------------------------------------------------------
// PlatformInput — yhteinen kantaluokka Android/iOS/Desktop
// ------------------------------------------------------------
// Toteuttaa kaikki IInput-metodit. Alustakohtaiset alaluokat
// ylikirjoittavat Update() ja syötteen keräyksen.
class PlatformInput : public IInput
{
public:
    PlatformInput()
        : m_quit(false)
        , m_key(STX_KEY_UNKNOWN)
        , m_keyPressed(STX_KEY_UNKNOWN)
        , m_keyReleased(STX_KEY_UNKNOWN)
        , m_mods(STX_KMOD_NONE)
        , m_mouseX(0), m_mouseY(0)
        , m_mouseState(0)
        , m_wheelY(0)
        , m_textInput("")
        , m_hasTextInput(false)
    {
        memset(m_keys, 0, sizeof(m_keys));
        memset(m_keysJustDown, 0, sizeof(m_keysJustDown));
        memset(m_keysJustUp, 0, sizeof(m_keysJustUp));
    }
    virtual ~PlatformInput() {}

    // --- IInput-rajapinta ---
    virtual bool Update() = 0;

    virtual bool IsKeyHeld(eKey key)      { return IsKeyDown(key); }
    virtual bool IsKeyDown(eKey key)
    {
        if ((unsigned)key >= kMaxKeys) return false;
        return m_keys[key] != 0;
    }
    virtual bool IsKeyUp(eKey key)        { return !IsKeyDown(key); }

    virtual bool OnShiftKeyPress()  { return (m_mods & STX_KMOD_SHIFT) != 0; }
    virtual bool OnLShiftKeyPress() { return (m_mods & STX_KMOD_LSHIFT) != 0; }
    virtual bool OnRShiftKeyPress() { return (m_mods & STX_KMOD_RSHIFT) != 0; }
    virtual bool OnCTRLKeyPress()   { return (m_mods & STX_KMOD_CTRL) != 0; }
    virtual bool OnLCTRLKeyPress()  { return (m_mods & STX_KMOD_LCTRL) != 0; }
    virtual bool OnRCTRLKeyPress()  { return (m_mods & STX_KMOD_RCTRL) != 0; }
    virtual bool OnALTKeyPress()    { return (m_mods & STX_KMOD_ALT) != 0; }
    virtual bool OnLALTKeyPress()   { return (m_mods & STX_KMOD_LALT) != 0; }
    virtual bool OnRALTKeyPress()   { return (m_mods & STX_KMOD_RALT) != 0; }
    virtual bool OnNUMKeyPress()    { return (m_mods & STX_KMOD_NUM) != 0; }
    virtual bool OnCAPSKeyPress()   { return (m_mods & STX_KMOD_CAPS) != 0; }
    virtual bool OnMODEKeyPress()   { return (m_mods & STX_KMOD_MODE) != 0; }

    virtual bool OnKeyPress(eKey key)   { return IsKeyDown(key); }
    virtual bool OnKeyRelease(eKey key) { return IsKeyUp(key); }
    virtual bool IsKeyPressed(eKey key) { return IsKeyDown(key); }
    virtual bool IsKeyPressed(eKey key, unsigned int) { return IsKeyDown(key); }

    virtual bool IsKeyPressed(void)  { return m_keyPressed != STX_KEY_UNKNOWN; }
    virtual bool IsKeyReleased(void) { return m_keyReleased != STX_KEY_UNKNOWN; }

    virtual bool IsWindowMoved(void)     { return false; }
    virtual bool IsWindowResized(void)   { return m_windowResized; }
    virtual bool IsWindowMinimised(void) { return m_windowMinimized; }
    virtual bool IsWindowMaximised(void) { return m_windowMaximized; }
    virtual bool IsWindowClosed(void)    { return m_windowClosed; }

    virtual bool IsMouseClick(void)          { return (m_mouseState & kMouseLeft) != 0; }
    virtual bool IsMouseDoubleClick(void)    { return false; }
    virtual bool IsMouseButtonPressed(void)  { return m_mouseState != 0; }
    virtual bool IsMouseButtonReleased(void) { return m_mouseState == 0; }
    virtual bool IsMouseButton(void)         { return m_mouseState != 0; }
    virtual bool IsMouseMotion(void)         { return m_mouseMoved; }
    virtual int  GetMouseX(void)             { return m_mouseX; }
    virtual int  GetMouseY(void)             { return m_mouseY; }

    virtual bool OnMouseButtonPress(eMouseButton b)   { return IsMouseButtonPressed(b); }
    virtual bool OnMouseButtonRelease(eMouseButton b) { return !IsMouseButtonPressed(b); }
    virtual bool IsMouseButtonPressed(eMouseButton b)
    {
        switch (b) {
            case STX_MBUTTON_LEFT:      return (m_mouseState & kMouseLeft)   != 0;
            case STX_MBUTTON_RIGHT:     return (m_mouseState & kMouseRight)  != 0;
            case STX_MBUTTON_MIDDLE:    return (m_mouseState & kMouseMiddle) != 0;
            case STX_MBUTTON_WHEELUP:   return m_wheelY > 0;
            case STX_MBUTTON_WHEELDOWN: return m_wheelY < 0;
        }
        return false;
    }
    virtual bool IsMouseButtonPressed(eMouseButton b, unsigned int) { return IsMouseButtonPressed(b); }

    // Joystick — oletuksena ei tukea (voi ylikirjoittaa)
    virtual int  GetJoyX()          { return 0; }
    virtual int  GetJoyY()          { return 0; }
    virtual bool IsJoyAxisMoving()  { return false; }
    virtual bool IsJoyBallMoving()  { return false; }
    virtual bool IsJoyHatMoving()   { return false; }
    virtual bool IsJoyButton()      { return false; }

    virtual Uint8  GetAxisJoystickDeviceIndex() { return 0; }
    virtual Uint8  GetAxisIndex()   { return 0; }
    virtual Sint16 GetAxisValue()   { return 0; }
    virtual Uint8  GetBallJoystickDeviceIndex() { return 0; }
    virtual Uint8  GetBallIndex()   { return 0; }
    virtual Sint16 GetBallXRel()    { return 0; }
    virtual Sint16 GetBallYRel()    { return 0; }
    virtual Uint8  GetHatJoystickDeviceIndex() { return 0; }
    virtual Uint8  GetHatIndex()    { return 0; }
    virtual Uint8  GetHatValue()    { return 0; }
    virtual Uint8  GetButtonJoystickDeviceIndex() { return 0; }
    virtual Uint8  GetButtonIndex() { return 0; }
    virtual Uint8  GetButtonState() { return 0; }

    virtual bool   Quit()          { return m_quit; }
    virtual int    getKeyID()      { return (int)m_keyPressed; }
    virtual char   getKeyChar()    { return (char)(m_keyPressed & 0xFF); }
    virtual Sint32 getKeyCode()    { return (Sint32)m_keyPressed; }

    // --- Kosketus (Android/iOS) ---
    virtual int   GetTouchCount()       { return (int)m_touches.size(); }
    virtual float GetTouchX(int idx)    { return (idx >= 0 && idx < (int)m_touches.size()) ? m_touches[idx].x : 0.0f; }
    virtual float GetTouchY(int idx)    { return (idx >= 0 && idx < (int)m_touches.size()) ? m_touches[idx].y : 0.0f; }
    virtual bool  IsTouchDown(int idx)  { return (idx >= 0 && idx < (int)m_touches.size()) && m_touches[idx].down; }
    virtual bool  IsTouchJustDown(int idx) { return (idx >= 0 && idx < (int)m_touches.size()) && m_touches[idx].justDown; }
    virtual bool  IsTouchJustUp(int idx)   { return (idx >= 0 && idx < (int)m_touches.size()) && m_touches[idx].justUp; }

    // --- Tekstinsyöttö ---
    virtual bool HasTextInput()            { return m_hasTextInput; }
    virtual const char* GetTextInput()     { return m_textInput.c_str(); }

    virtual bool Init()     { return true; }
    virtual void Shutdown() {}

protected:
    static const unsigned kMaxKeys = 512;
    struct Touch {
        float x, y;
        bool down, justDown, justUp;
        int  id;
    };

    unsigned char m_keys[kMaxKeys];
    unsigned char m_keysJustDown[kMaxKeys];
    unsigned char m_keysJustUp[kMaxKeys];

    eKey  m_key, m_keyPressed, m_keyReleased;
    STX_Keymod m_mods;

    int    m_mouseX, m_mouseY;
    unsigned int m_mouseState;   // bitmask
    int    m_wheelY;
    bool   m_mouseMoved;

    bool m_windowResized, m_windowMinimized, m_windowMaximized, m_windowClosed;
    bool m_quit;

    std::vector<Touch> m_touches;
    std::string m_textInput;
    bool m_hasTextInput;

    static const unsigned int kMouseLeft   = 1u << 0;
    static const unsigned int kMouseRight  = 1u << 1;
    static const unsigned int kMouseMiddle = 1u << 2;

    // Apufunktiot alaluokille
    void ClearFrame()
    {
        memset(m_keysJustDown, 0, sizeof(m_keysJustDown));
        memset(m_keysJustUp,   0, sizeof(m_keysJustUp));
        m_keyPressed  = STX_KEY_UNKNOWN;
        m_keyReleased = STX_KEY_UNKNOWN;
        m_wheelY      = 0;
        m_mouseMoved  = false;
        m_windowResized = false;
        m_windowMinimized = false;
        m_windowMaximized = false;
        m_windowClosed = false;
        m_hasTextInput = false;
        m_textInput.clear();
        for (size_t i = 0; i < m_touches.size(); ++i) {
            m_touches[i].justDown = false;
            m_touches[i].justUp   = false;
        }
    }

    void SetKeyDown(eKey k, bool down)
    {
        if ((unsigned)k >= kMaxKeys) return;
        if (down) {
            if (!m_keys[k]) m_keysJustDown[k] = 1;
            m_keys[k] = 1;
            m_keyPressed = k;
        } else {
            if (m_keys[k]) m_keysJustUp[k] = 1;
            m_keys[k] = 0;
            m_keyReleased = k;
        }
    }

    void AddTextInput(const char* utf8)
    {
        if (!utf8 || !*utf8) return;
        m_textInput += utf8;
        m_hasTextInput = true;
    }

    Touch* FindTouch(int id)
    {
        for (size_t i = 0; i < m_touches.size(); ++i)
            if (m_touches[i].id == id) return &m_touches[i];
        return NULL;
    }
    Touch* AddTouch(int id)
    {
        Touch t; t.id = id; t.down = true; t.justDown = true;
        t.justUp = false; t.x = t.y = 0.0f;
        m_touches.push_back(t);
        return &m_touches.back();
    }
    void RemoveTouch(int id)
    {
        for (size_t i = 0; i < m_touches.size(); ++i) {
            if (m_touches[i].id == id) {
                m_touches[i].down = false;
                m_touches[i].justUp = true;
                return;
            }
        }
    }
};

// Android-näppäinkoodien käännös eKey:ksi
static eKey AndroidKeyToSTX(int32_t keycode, int32_t metaState)
{
    // Erityisnäppäimet
    switch (keycode) {
        case AKEYCODE_BACK:        return STX_KEY_ESCAPE; // "back" = escape
        case AKEYCODE_ENTER:       return STX_KEY_RETURN;
        case AKEYCODE_DEL:         return STX_KEY_BACKSPACE;
        case AKEYCODE_TAB:         return STX_KEY_TAB;
        case AKEYCODE_SPACE:       return STX_KEY_SPACE;
        case AKEYCODE_ESCAPE:      return STX_KEY_ESCAPE;
        case AKEYCODE_DPAD_UP:     return STX_KEY_UP;
        case AKEYCODE_DPAD_DOWN:   return STX_KEY_DOWN;
        case AKEYCODE_DPAD_LEFT:   return STX_KEY_LEFT;
        case AKEYCODE_DPAD_RIGHT:  return STX_KEY_RIGHT;
        case AKEYCODE_MOVE_HOME:   return STX_KEY_HOME;
        case AKEYCODE_MOVE_END:    return STX_KEY_END;
        case AKEYCODE_PAGE_UP:     return STX_KEY_PAGEUP;
        case AKEYCODE_PAGE_DOWN:   return STX_KEY_PAGEDOWN;
        case AKEYCODE_INSERT:      return STX_KEY_INSERT;
        case AKEYCODE_FORWARD_DEL: return STX_KEY_DELETE;
        case AKEYCODE_CAPS_LOCK:   return STX_KEY_CAPSLOCK;
        case AKEYCODE_SHIFT_LEFT:  return STX_KEY_LSHIFT;
        case AKEYCODE_SHIFT_RIGHT: return STX_KEY_RSHIFT;
        case AKEYCODE_CTRL_LEFT:   return STX_KEY_LCTRL;
        case AKEYCODE_CTRL_RIGHT:  return STX_KEY_RCTRL;
        case AKEYCODE_ALT_LEFT:    return STX_KEY_LALT;
        case AKEYCODE_ALT_RIGHT:   return STX_KEY_RALT;
        case AKEYCODE_MENU:        return STX_KEY_MENU;
        case AKEYCODE_NUM_LOCK:    return STX_KEY_NUMLOCK;
        case AKEYCODE_BREAK:       return STX_KEY_BREAK;
        case AKEYCODE_SYSRQ:       return STX_KEY_PRINT;
        case AKEYCODE_HELP:        return STX_KEY_HELP;
        // Numeric keypad
        case AKEYCODE_NUMPAD_0:      return STX_KEY_KP0;
        case AKEYCODE_NUMPAD_1:      return STX_KEY_KP1;
        case AKEYCODE_NUMPAD_2:      return STX_KEY_KP2;
        case AKEYCODE_NUMPAD_3:      return STX_KEY_KP3;
        case AKEYCODE_NUMPAD_4:      return STX_KEY_KP4;
        case AKEYCODE_NUMPAD_5:      return STX_KEY_KP5;
        case AKEYCODE_NUMPAD_6:      return STX_KEY_KP6;
        case AKEYCODE_NUMPAD_7:      return STX_KEY_KP7;
        case AKEYCODE_NUMPAD_8:      return STX_KEY_KP8;
        case AKEYCODE_NUMPAD_9:      return STX_KEY_KP9;
        case AKEYCODE_NUMPAD_DOT:    return STX_KEY_KP_PERIOD;
        case AKEYCODE_NUMPAD_DIVIDE: return STX_KEY_KP_DIVIDE;
        case AKEYCODE_NUMPAD_MULTIPLY: return STX_KEY_KP_MULTIPLY;
        case AKEYCODE_NUMPAD_SUBTRACT: return STX_KEY_KP_MINUS;
        case AKEYCODE_NUMPAD_ADD:      return STX_KEY_KP_PLUS;
        case AKEYCODE_NUMPAD_ENTER:    return STX_KEY_KP_ENTER;
        case AKEYCODE_NUMPAD_EQUALS:   return STX_KEY_KP_EQUALS;
        // F-näppäimet
        case AKEYCODE_F1: return STX_KEY_F1;  case AKEYCODE_F2: return STX_KEY_F2;
        case AKEYCODE_F3: return STX_KEY_F3;  case AKEYCODE_F4: return STX_KEY_F4;
        case AKEYCODE_F5: return STX_KEY_F5;  case AKEYCODE_F6: return STX_KEY_F6;
        case AKEYCODE_F7: return STX_KEY_F7;  case AKEYCODE_F8: return STX_KEY_F8;
        case AKEYCODE_F9: return STX_KEY_F9;  case AKEYCODE_F10: return STX_KEY_F10;
        case AKEYCODE_F11: return STX_KEY_F11; case AKEYCODE_F12: return STX_KEY_F12;
    }

    // Kirjaimet: AKEYCODE_A = 29 ... AKEYCODE_Z = 54 (25 kpl)
    if (keycode >= AKEYCODE_A && keycode <= AKEYCODE_Z) {
        bool shift = (metaState & (AMETA_SHIFT_ON | AMETA_SHIFT_LEFT_ON |
                                   AMETA_SHIFT_RIGHT_ON)) != 0;
        char c = (char)('a' + (keycode - AKEYCODE_A));
        if (shift) c = (char)('A' + (keycode - AKEYCODE_A));
        // Palauta vastaava eKey (a...z tai A...Z; STX_KEY_a...z)
        return (eKey)((int)STX_KEY_a + (keycode - AKEYCODE_A));
    }

    // Numerot: AKEYCODE_0 = 7 ... AKEYCODE_9 = 16
    if (keycode >= AKEYCODE_0 && keycode <= AKEYCODE_9) {
        return (eKey)((int)STX_KEY_0 + (keycode - AKEYCODE_0));
    }

    // Välimerkit
    switch (keycode) {
        case AKEYCODE_COMMA:      return STX_KEY_COMMA;
        case AKEYCODE_PERIOD:     return STX_KEY_PERIOD;
        case AKEYCODE_SLASH:      return STX_KEY_SLASH;
        case AKEYCODE_SEMICOLON:  return STX_KEY_SEMICOLON;
        case AKEYCODE_APOSTROPHE: return STX_KEY_QUOTE;
        case AKEYCODE_LEFT_BRACKET:  return STX_KEY_LEFTBRACKET;
        case AKEYCODE_RIGHT_BRACKET: return STX_KEY_RIGHTBRACKET;
        case AKEYCODE_BACKSLASH:  return STX_KEY_BACKSLASH;
        case AKEYCODE_GRAVE:      return STX_KEY_BACKQUOTE;
        case AKEYCODE_MINUS:      return STX_KEY_MINUS;
        case AKEYCODE_EQUALS:     return STX_KEY_EQUALS;
        case AKEYCODE_PLUS:       return STX_KEY_PLUS;
    }

    return STX_KEY_UNKNOWN;
}

static STX_Keymod AndroidMetaToSTX(int32_t metaState)
{
    STX_Keymod m = STX_KMOD_NONE;
    if (metaState & AMETA_SHIFT_LEFT_ON)  m = (STX_Keymod)(m | STX_KMOD_LSHIFT);
    if (metaState & AMETA_SHIFT_RIGHT_ON) m = (STX_Keymod)(m | STX_KMOD_RSHIFT);
    if (metaState & (AMETA_SHIFT_ON | AMETA_SHIFT_LEFT_ON | AMETA_SHIFT_RIGHT_ON))
        m = (STX_Keymod)(m | STX_KMOD_SHIFT);
    if (metaState & AMETA_CTRL_LEFT_ON)   m = (STX_Keymod)(m | STX_KMOD_LCTRL);
    if (metaState & AMETA_CTRL_RIGHT_ON)  m = (STX_Keymod)(m | STX_KMOD_RCTRL);
    if (metaState & (AMETA_CTRL_ON | AMETA_CTRL_LEFT_ON | AMETA_CTRL_RIGHT_ON))
        m = (STX_Keymod)(m | STX_KMOD_CTRL);
    if (metaState & AMETA_ALT_LEFT_ON)    m = (STX_Keymod)(m | STX_KMOD_LALT);
    if (metaState & AMETA_ALT_RIGHT_ON)   m = (STX_Keymod)(m | STX_KMOD_RALT);
    if (metaState & (AMETA_ALT_ON | AMETA_ALT_LEFT_ON | AMETA_ALT_RIGHT_ON))
        m = (STX_Keymod)(m | STX_KMOD_ALT);
    if (metaState & AMETA_CAPS_LOCK_ON)   m = (STX_Keymod)(m | STX_KMOD_CAPS);
    if (metaState & AMETA_NUM_LOCK_ON)    m = (STX_Keymod)(m | STX_KMOD_NUM);
    if (metaState & AMETA_META_LEFT_ON)   m = (STX_Keymod)(m | STX_KMOD_LGUI);
    if (metaState & AMETA_META_RIGHT_ON)  m = (STX_Keymod)(m | STX_KMOD_RGUI);
    if (metaState & (AMETA_META_ON | AMETA_META_LEFT_ON | AMETA_META_RIGHT_ON))
        m = (STX_Keymod)(m | STX_KMOD_GUI);
    return m;
}

class AndroidInput : public PlatformInput
{
public:
    AndroidInput() : PlatformInput() {}

    // Kutsutaan JNI:sta (Java-puolelta) — ks. AndroidInputBridge.java
    void OnNativeKeyEvent(int action, int keycode, int metaState)
    {
        eKey k = AndroidKeyToSTX(keycode, metaState);
        if (k == STX_KEY_UNKNOWN) return;
        m_mods = AndroidMetaToSTX(metaState);
        if (action == AKEY_EVENT_ACTION_DOWN) {
            SetKeyDown(k, true);
        } else if (action == AKEY_EVENT_ACTION_UP) {
            SetKeyDown(k, false);
        } else if (action == AKEY_EVENT_ACTION_MULTIPLE) {
            SetKeyDown(k, true);
            SetKeyDown(k, false);
        }
    }

    void OnNativeTouchEvent(int action, int pointerIndex, int pointerId,
                            float x, float y)
    {
        // action on ACTION_DOWN/MOVE/UP/CANCEL/POINTER_DOWN/POINTER_UP
        const int ACTION_MASK = 0xff;
        const int ACTION_DOWN = 0, ACTION_UP = 1, ACTION_MOVE = 2,
                  ACTION_CANCEL = 3, ACTION_POINTER_DOWN = 5,
                  ACTION_POINTER_UP = 6;

        int a = action & ACTION_MASK;
        if (a == ACTION_DOWN || a == ACTION_POINTER_DOWN) {
            Touch* t = FindTouch(pointerId);
            if (!t) t = AddTouch(pointerId);
            if (t) { t->x = x; t->y = y; t->down = true; t->justDown = true; t->justUp = false; }
            m_mouseX = (int)x; m_mouseY = (int)y;
            m_mouseState |= kMouseLeft;
            m_mouseMoved = true;
        } else if (a == ACTION_MOVE) {
            Touch* t = FindTouch(pointerId);
            if (t) { t->x = x; t->y = y; }
            m_mouseX = (int)x; m_mouseY = (int)y;
            m_mouseMoved = true;
        } else if (a == ACTION_UP || a == ACTION_POINTER_UP) {
            Touch* t = FindTouch(pointerId);
            if (t) { t->x = x; t->y = y; t->down = false; t->justUp = true; }
            if (m_touches.empty()) m_mouseState &= ~kMouseLeft;
            m_mouseMoved = true;
        } else if (a == ACTION_CANCEL) {
            for (size_t i = 0; i < m_touches.size(); ++i) {
                m_touches[i].down = false; m_touches[i].justUp = true;
            }
            m_mouseState = 0;
        }
    }

    void OnNativeText(const char* utf8) { AddTextInput(utf8); }

    virtual bool Update()
    {
        // Android: JNI-kutsut päivittävät tilan suoraan (tapahtumaohjattu).
        // Tässä vain "frame"-tilan nollaus.
        ClearFrame();
        return m_quit;
    }

    virtual bool Init() { return true; }
};

}

// iOS-otsikot — lisää tarvittaessa #import <UIKit/UIKit.h> .mm-tiedostossa
// Tässä .cpp:ssä käytetään vain C-rajapintaa; UIKit-kytkentä tehdään .mm:ssä
// ja kutsutaan näitä C-funktioita.

class IosInput : public PlatformInput
{
public:
    IosInput() : PlatformInput() {}

    // C-rajapinta, jota .mm kutsuu
    void OnNativeTouch(int phase, int touchId, float x, float y)
    {
        // phase: 0=began, 1=moved, 2=ended, 3=cancelled
        if (phase == 0) {
            Touch* t = FindTouch(touchId);
            if (!t) t = AddTouch(touchId);
            if (t) { t->x = x; t->y = y; t->down = true; t->justDown = true; }
            m_mouseX = (int)x; m_mouseY = (int)y;
            m_mouseState |= kMouseLeft;
            m_mouseMoved = true;
        } else if (phase == 1) {
            Touch* t = FindTouch(touchId);
            if (t) { t->x = x; t->y = y; }
            m_mouseX = (int)x; m_mouseY = (int)y;
            m_mouseMoved = true;
        } else if (phase == 2 || phase == 3) {
            Touch* t = FindTouch(touchId);
            if (t) { t->x = x; t->y = y; t->down = false; t->justUp = true; }
            if (m_touches.empty()) m_mouseState &= ~kMouseLeft;
            m_mouseMoved = true;
        }
    }

    void OnNativeKey(int down, int keycode, int mods)
    {
        // keycode: iOS UIKeyboardHIDUsage -koodi, mods: STX_Keymod-yhteensopiva
        eKey k = IosKeyToSTX(keycode);
        if (k == STX_KEY_UNKNOWN) return;
        m_mods = (STX_Keymod)mods;
        SetKeyDown(k, down != 0);
    }

    void OnNativeText(const char* utf8) { AddTextInput(utf8); }

    virtual bool Update() { ClearFrame(); return m_quit; }

private:
    static eKey IosKeyToSTX(int hid)
    {
        // HID Usage Tables (Keyboard/Keypad), https://www.usb.org/hid
        switch (hid) {
            case 0x29: return STX_KEY_ESCAPE;
            case 0x28: return STX_KEY_RETURN;
            case 0x2A: return STX_KEY_BACKSPACE;
            case 0x2B: return STX_KEY_TAB;
            case 0x2C: return STX_KEY_SPACE;
            case 0x4F: return STX_KEY_RIGHT;
            case 0x50: return STX_KEY_LEFT;
            case 0x51: return STX_KEY_DOWN;
            case 0x52: return STX_KEY_UP;

// Tässä käytetään SDL:n SDLK_*-arvoja, jotka ovat jo eKey-enumin arvoja.

// SDL:n scancode -> eKey -käännös (tarvitaan vain jos halutaan)

static STX_Keymod SDLModToSTX(Uint16 mod)
{
    STX_Keymod m = STX_KMOD_NONE;
    if (mod & KMOD_LSHIFT) m = (STX_Keymod)(m | STX_KMOD_LSHIFT);
    if (mod & KMOD_RSHIFT) m = (STX_Keymod)(m | STX_KMOD_RSHIFT);
    if (mod & KMOD_LCTRL)  m = (STX_Keymod)(m | STX_KMOD_LCTRL);
    if (mod & KMOD_RCTRL)  m = (STX_Keymod)(m | STX_KMOD_RCTRL);
    if (mod & KMOD_LALT)   m = (STX_Keymod)(m | STX_KMOD_LALT);
    if (mod & KMOD_RALT)   m = (STX_Keymod)(m | STX_KMOD_RALT);
    if (mod & KMOD_LGUI)   m = (STX_Keymod)(m | STX_KMOD_LGUI);
    if (mod & KMOD_RGUI)   m = (STX_Keymod)(m | STX_KMOD_RGUI);
    if (mod & KMOD_NUM)    m = (STX_Keymod)(m | STX_KMOD_NUM);
    if (mod & KMOD_CAPS)   m = (STX_Keymod)(m | STX_KMOD_CAPS);
    if (mod & KMOD_MODE)   m = (STX_Keymod)(m | STX_KMOD_MODE);
    return m;
}

class SDLInputImpl : public PlatformInput
{
public:
    SDLInputImpl() : PlatformInput()
    {
        m_keyboardState = SDL_GetKeyboardState(NULL);
        m_mouseState = SDL_GetMouseState(&m_mouseX, &m_mouseY);
    }
    virtual ~SDLInputImpl() {}

    virtual bool Update()
    {
        ClearFrame();
        SDL_Event event;
        while (SDL_PollEvent(&event) != 0) {
            switch (event.type) {
#if defined(SDL_2_0_5_BUILD)
                case SDL_WINDOWEVENT:
                    if (event.window.event == SDL_WINDOWEVENT_RESIZED)
                        m_windowResized = true;
                    if (event.window.event == SDL_WINDOWEVENT_MINIMIZED)
                        m_windowMinimized = true;
                    if (event.window.event == SDL_WINDOWEVENT_MAXIMIZED)
                        m_windowMaximized = true;
                    if (event.window.event == SDL_WINDOWEVENT_MOVED)
                        { /* moved */ }
                    if (event.window.event == SDL_WINDOWEVENT_CLOSE)
                        m_windowClosed = true;
                    break;
#else
                case SDL_VIDEORESIZE:
                    m_windowResized = true;
                    break;
#endif
                case SDL_QUIT:
                    m_quit = true;
                    return true;
                case SDL_KEYDOWN: {
                    eKey k = SDLKtoKEY(event.key.keysym.sym);
                    if (k == STX_KEY_ESCAPE) { m_quit = true; return true; }
                    m_mods = SDLModToSTX((Uint16)event.key.keysym.mod);
                    SetKeyDown(k, true);
                    // Tekstinsyöttö (SDL2)
#if defined(SDL_2_0_5_BUILD)
                    if (event.key.keysym.sym >= 32 && event.key.keysym.sym < 127) {
                        char buf[2] = { (char)event.key.keysym.sym, 0 };
                        AddTextInput(buf);
                    }
#endif
                    break;
                }
                case SDL_KEYUP: {
                    eKey k = SDLKtoKEY(event.key.keysym.sym);
                    m_mods = SDLModToSTX((Uint16)event.key.keysym.mod);
                    SetKeyDown(k, false);
                    break;
                }
                case SDL_MOUSEMOTION:
                    m_mouseX = event.motion.x;
                    m_mouseY = event.motion.y;
                    m_mouseMoved = true;
                    break;
                case SDL_MOUSEBUTTONDOWN:
                    m_mouseX = event.button.x;
                    m_mouseY = event.button.y;
                    if (event.button.button == SDL_BUTTON_LEFT)   m_mouseState |= kMouseLeft;
                    if (event.button.button == SDL_BUTTON_RIGHT)  m_mouseState |= kMouseRight;
                    if (event.button.button == SDL_BUTTON_MIDDLE) m_mouseState |= kMouseMiddle;
                    m_mouseMoved = true;
                    break;
                case SDL_MOUSEBUTTONUP:
                    m_mouseX = event.button.x;
                    m_mouseY = event.button.y;
                    if (event.button.button == SDL_BUTTON_LEFT)   m_mouseState &= ~kMouseLeft;
                    if (event.button.button == SDL_BUTTON_RIGHT)  m_mouseState &= ~kMouseRight;
                    if (event.button.button == SDL_BUTTON_MIDDLE) m_mouseState &= ~kMouseMiddle;
                    m_mouseMoved = true;
                    break;
#if defined(SDL_2_0_5_BUILD)
                case SDL_MOUSEWHEEL:
                    m_wheelY = event.wheel.y;
                    m_iWheelY = event.wheel.y;
                    break;
#else
                case SDL_MOUSEBUTTONDOWN: // (jo käsitelty; wheel SDL1.2:na)
                    break;
#endif
                default:
                    break;
            }
        }

        // Päivitä spin-arvot hiiren sijainnin perusteella
        int w = 0, h = 0;
#if defined(SDL_2_0_5_BUILD)
        SDL_Window* win = STX_Service::GetSDLWindow();
        if (win) SDL_GetWindowSize(win, &w, &h);
#else
        const SDL_VideoInfo* vi = SDL_GetVideoInfo();
        if (vi) { w = vi->current_w; h = vi->current_h; }
#endif
        float fx = 0, fy = 0;
        STX_UpdateSpinFromMouse((float)m_mouseX, (float)m_mouseY,
                                (float)w, (float)h, fx, fy);
        m_fX = fx; m_fY = fy;
        if (IsMouseButtonPressed(STX_MBUTTON_LEFT)) {
            m_fSpinX = fx; m_fSpinY = fy;
            m_fSpinX_L = fx; m_fSpinY_L = fy;
        }
        if (IsMouseButtonPressed(STX_MBUTTON_RIGHT)) {
            m_fSpinX_R = fx; m_fSpinY_R = fy;
        }
        if (IsMouseButtonPressed(STX_MBUTTON_WHEELDOWN))
            m_fZ = fy;

        return false;
    }

    virtual bool OnShiftKeyPress()  { return (m_mods & STX_KMOD_SHIFT) != 0; }
    virtual bool OnLShiftKeyPress() { return (m_mods & STX_KMOD_LSHIFT) != 0; }
    virtual bool OnRShiftKeyPress() { return (m_mods & STX_KMOD_RSHIFT) != 0; }
    virtual bool OnCTRLKeyPress()   { return (m_mods & STX_KMOD_CTRL) != 0; }
    virtual bool OnLCTRLKeyPress()  { return (m_mods & STX_KMOD_LCTRL) != 0; }
    virtual bool OnRCTRLKeyPress()  { return (m_mods & STX_KMOD_RCTRL) != 0; }
    virtual bool OnALTKeyPress()    { return (m_mods & STX_KMOD_ALT) != 0; }
    virtual bool OnLALTKeyPress()   { return (m_mods & STX_KMOD_LALT) != 0; }
    virtual bool OnRALTKeyPress()   { return (m_mods & STX_KMOD_RALT) != 0; }
    virtual bool OnNUMKeyPress()    { return (m_mods & STX_KMOD_NUM) != 0; }
    virtual bool OnCAPSKeyPress()   { return (m_mods & STX_KMOD_CAPS) != 0; }
    virtual bool OnMODEKeyPress()   { return (m_mods & STX_KMOD_MODE) != 0; }

private:
    const Uint8* m_keyboardState;
};
```

### Android — `InputBridge.java`

```java
package fi.tommironty.STX;

import android.app.Activity;
import android.view.KeyEvent;
import android.view.MotionEvent;
import android.view.View;
import android.view.inputmethod.InputMethodManager;
import android.content.Context;

public class InputBridge {
    static {
        System.loadLibrary("stxinput");
    }
    private static native void nativeKeyEvent(int action, int keycode, int metaState);
    private static native void nativeTouchEvent(int action, int pointerIndex,
                                                int pointerId, float x, float y);
    private static native void nativeTextInput(String text);

    // Asenna Activityyn:
    public static void install(final Activity activity) {
        View v = activity.getWindow().getDecorView();
        v.setOnKeyListener(new View.OnKeyListener() {
            @Override public boolean onKey(View v, int keyCode, KeyEvent event) {
                nativeKeyEvent(event.getAction(), keyCode, event.getMetaState());
                return false;
            }
        });
        v.setOnTouchListener(new View.OnTouchListener() {
            @Override public boolean onTouch(View v, MotionEvent ev) {
                int action = ev.getActionMasked();
                int idx = ev.getActionIndex();
                int id = ev.getPointerId(idx);
                float x = ev.getX(idx), y = ev.getY(idx);
                nativeTouchEvent(action, idx, id, x, y);
                return false;
            }
        });
    }
}
```

## Yhteenveto muutoksista

| Alue | Ennen | Jälkeen |
|------|-------|---------|
| **eKey-enum** | Sidottu `SDLK_*`-makroon | `STX_KEYVAL()`-makro valitsee SDL:n tai itsenäisen arvon |
| **STX_Keymod** | Sidottu `KMOD_*`-vakioihin | Itsenäinen enum, SDL-yhteensopivin arvoin |
| **eKeyButton, eJoyButton, eMouseEvent, eMouseButton** | Osa SDL:ää | Itsenäisiä |
| **IInput** | Oli jo alustariippumaton | Lisätty kosketus- ja tekstinsyöttömetodit (oletustoteutus) |
| **Toteutus** | Vain SDLInput + DummyInput | `PlatformInput`-kanta + `AndroidInput`, `IosInput`, `SDLInputImpl`, `GenericDummyInput` |
| **Factory** | SDL/Dummy-valinta | Valitsee oikean alustan `#if defined(__ANDROID__)` / `__APPLE__` / `SDL_*` |

**Keskeiset edut:**
1. Sama `IInput`-rajapinta kaikilla alustoilla → pelikoodi ei muutu.
2. Android/iOS eivät tarvitse SDL:ää → kevyempi ja helpompi integroida GLES2-renderöintiin.
3. SDL2-toteutus säilyy työpöydällä (Linux/Windows/OSX).
4. `eKey`-arvojen numeeriset vakiot ovat SDL1.2-yhteensopivia, joten vanha koodi toimii sellaisenaan.
5. Kosketus ja monikosketus (multi-touch) tuetaan Androidilla ja iOS:llä.
6. Tekstinsyöttö (IME) tuetaan kaikilla alustoilla.
